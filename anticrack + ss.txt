        // Get Local IP (Private IP)
        private static string GetLocalIPAddress()
        {
            var host = Dns.GetHostEntry(Dns.GetHostName());
            foreach (var ip in host.AddressList)
            {
                if (ip.AddressFamily == System.Net.Sockets.AddressFamily.InterNetwork)
                {
                    return ip.ToString();
                }
            }
            return "Not Found";
        }

        // Get Public IP
        private static string GetPublicIP()
        {
            try
            {
                using (WebClient client = new WebClient())
                {
                    return client.DownloadString("https://api.ipify.org");
                }
            }
            catch
            {
                return "Unable to fetch";
            }
        }

        // Get HWID (Hardware ID)
        private static string GetHWID()
        {
            return WindowsIdentity.GetCurrent().User.Value;
        }

        // Get Motherboard ID using WMI
        private static string GetMotherboardID()
        {
            try
            {
                ManagementObjectSearcher searcher = new ManagementObjectSearcher("SELECT * FROM Win32_BaseBoard");
                foreach (ManagementObject wmi in searcher.Get())
                {
                    try
                    {
                        return wmi.GetPropertyValue("SerialNumber").ToString();
                    }
                    catch { }
                }
            }
            catch { }
            return "Unknown";
        }

        // Get MD5 Hash of the application or file
        private static string GetMD5Hash(string filePath)
        {
            using (var md5 = MD5.Create())
            {
                using (var stream = File.OpenRead(filePath))
                {
                    return BitConverter.ToString(md5.ComputeHash(stream)).Replace("-", "").ToLower();
                }
            }
        }

        // Get the computer name
        private static string GetComputerName()
        {
            return Environment.MachineName;
        }

        // Detect if a process is running
        private static bool iscrackeraprunring11(string processName)

        //send ss
        private static string CaptureScreenshot()
        {
            string screenshotPath = Path.Combine(Path.GetTempPath(), "screenshot.png");

            Bitmap bitmap = new Bitmap(Screen.PrimaryScreen.Bounds.Width, Screen.PrimaryScreen.Bounds.Height);
            using (Graphics g = Graphics.FromImage(bitmap))
            {
                g.CopyFromScreen(0, 0, 0, 0, bitmap.Size);
            }

            bitmap.Save(screenshotPath, ImageFormat.Png);
            return screenshotPath;
        }
